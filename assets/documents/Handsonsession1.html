# Cell 13 - President with Highest Average Word Length
average_length = {}

for file in inaugural.fileids():

    words = [
        word
        for word in inaugural.words(file)
        if word.isalpha()
    ]

    avg = sum(len(word) for word in words) / len(words)

    average_length[file] = avg

best = max(average_length, key=average_length.get)

print("President with Highest Average Word Length")
print("------------------------------------------")
print(best)
print("Average Word Length :", round(average_length[best], 2))
